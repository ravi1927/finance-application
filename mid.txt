function CollectionModal({session,customers,setCustomers,setCollections,close}:any){
  const allowed=customers.filter((c:Customer)=>session.role==='manager'||(c.areaId===session.areaId&&c.agentIds.includes(session.id)));
  const [cid,setCid]=useState(allowed[0]?.id||'');
  const [amount,setAmount]=useState('');
  const [method,setMethod]=useState<'Cash'|'UPI'|'Bank'>('Cash');
  const [referenceNo,setReferenceNo]=useState('');
  const [proofUri,setProofUri]=useState<string|null>(null);
  const [busy,setBusy]=useState(false);
  const proofKind=method==='Cash'?'Customer confirmation photo':method==='UPI'?'UPI screenshot':'Bank transaction proof';
  const resetProof=()=>{setProofUri(null);setReferenceNo('')};
  const takePhoto=async()=>{
    const permission=await ImagePicker.requestCameraPermissionsAsync();
    if(!permission.granted){Alert.alert('Camera permission required','Allow camera access to capture the collection proof.');return}
    const result=await ImagePicker.launchCameraAsync({mediaTypes:['images'],allowsEditing:true,quality:.75});
    if(!result.canceled)setProofUri(result.assets[0].uri);
  };
  const choosePhoto=async()=>{
    const permission=await ImagePicker.requestMediaLibraryPermissionsAsync();
    if(!permission.granted){Alert.alert('Photo permission required','Allow photo access to select the collection proof.');return}
    const result=await ImagePicker.launchImageLibraryAsync({mediaTypes:['images'],allowsEditing:true,quality:.75});
    if(!result.canceled)setProofUri(result.assets[0].uri);
  };
  const save=()=>{
    const n=Number(amount);const c=customers.find((x:Customer)=>x.id===cid);
    if(!c||!n||n<=0){Alert.alert('Check amount','Enter a valid collection amount.');return}
    if(n>c.outstanding){Alert.alert('Amount too high','Collection cannot exceed current outstanding.');return}
    if(!proofUri){Alert.alert('Proof required',`${proofKind} is required before this collection can be confirmed.`);return}
    if((method==='UPI'||method==='Bank')&&!referenceNo.trim()){Alert.alert('Reference required',`${method==='UPI'?'UTR':'Bank reference / UTR'} is required.`);return}
    setBusy(true);
    const t=now();const receipt=`FIN-${Date.now().toString().slice(-7)}`;
    const item:Collection={id:receipt,customerId:c.id,agentId:session.role==='boy'?session.id:'M001',areaId:c.areaId,amount:n,principal:Math.max(0,n-Math.round(n*.1)),interest:Math.min(n,Math.round(n*.1)),date:t.date,time:t.time,method,type:'Loan',login:session.id,frequency:c.frequency,status:'Confirmed',proofUri,proofKind,referenceNo:referenceNo.trim()||undefined,collectorConfirmedAt:new Date().toISOString(),managerVerified:false};
    setCollections((old:Collection[])=>[item,...old]);
    setCustomers((old:Customer[])=>old.map(x=>x.id===c.id?{...x,outstanding:x.outstanding-n}:x));
    setBusy(false);close();Alert.alert('Payment confirmed',`Receipt ${receipt} created. The customer can now see the payment slip.`)
  };
  return <Modal visible transparent animationType="slide" onRequestClose={close}><Pressable style={s.modalBg} onPress={close}><View style={s.formSheet} onStartShouldSetResponder={()=>true}>
    <Text style={s.drawerTitle}>Confirm collection</Text><Text style={s.drawerSub}>Required proof is stored against the same financial transaction.</Text>
    <ScrollView keyboardShouldPersistTaps="handled">
      {allowed.slice(0,8).map((c:Customer)=><Pressable key={c.id} onPress={()=>{setCid(c.id);resetProof()}} style={[s.customerPick,c.id===cid&&{borderColor:C.blue,backgroundColor:C.softBlue}]}><View style={{flex:1}}><Text style={s.bold}>{c.name}</Text><Text style={s.muted}>Outstanding {money(c.outstanding)} • Due {money(c.dueAmount)}</Text></View><Text style={{color:c.id===cid?C.blue:C.muted,fontWeight:'900'}}>✓</Text></Pressable>)}
      <Input label="Amount collected" value={amount} onChangeText={setAmount} placeholder="₹ 0" keyboardType="numeric"/>
      <Text style={s.label}>Payment method</Text><View style={s.methodRow}>{(['Cash','UPI','Bank'] as const).map(x=><Pill key={x} text={x} active={method===x} onPress={()=>{setMethod(x);setReferenceNo('');setProofUri(null)}}/>)}</View>
      {method!=='Cash'&&<Input label={method==='UPI'?'UTR / UPI transaction ID':'Bank reference / UTR'} value={referenceNo} onChangeText={setReferenceNo} placeholder={method==='UPI'?'e.g. 123456789012':'e.g. UTR987654321'} autoCapitalize="characters"/>}
      <Card style={s.proofCard}><Text style={s.bold}>{proofKind}</Text><Text style={s.muted}>{method==='Cash'?'Capture the customer confirmation/signature photo.':method==='UPI'?'Capture or select the successful UPI payment screenshot.':'Capture or select the bank transfer confirmation/proof.'}</Text>{proofUri?<View style={{marginTop:12}}><Image source={{uri:proofUri}} style={s.proofPreview}/><View style={{flexDirection:'row',gap:8,marginTop:8}}><Button title="Retake / replace" secondary onPress={takePhoto}/><Button title="Remove" secondary onPress={()=>setProofUri(null)}/></View></View>:<View style={{flexDirection:'row',gap:8,marginTop:12}}><Button title="Take photo" onPress={takePhoto}/><Button title="Choose photo" secondary onPress={choosePhoto}/></View>}</Card>
      <View style={s.proofRule}><Text style={s.proofRuleTitle}>Before confirmation</Text><Text style={s.muted}>{method==='Cash'?'Cash requires customer confirmation photo.':method==='UPI'?'UPI requires both UTR and payment screenshot.':'Bank requires both reference/UTR and transaction proof.'}</Text></View>
      <Button title={busy?'Saving…':'Confirm & issue receipt'} onPress={save} disabled={busy}/>
    </ScrollView>
  </View></Pressable></Modal>
}
function ExpenseModal({session,setExpenses,close}:any){const [type,setType]=useState<'Food'|'Petrol'|'Travel'|'Other'>('Petrol');const [amount,setAmount]=useState('');const [note,setNote]=useState('');const save=()=>{const n=Number(amount);if(!n)return;const t=now();setExpenses((old:Expense[])=>[{id:`EXP-${Date.now()}`,areaId:session.areaId,agentId:session.id,type,amount:n,date:t.date,note:note||'Field expense'},...old]);close()};return <Modal visible transparent animationType="slide"><Pressable style={s.modalBg} onPress={close}><View style={s.formSheet} onStartShouldSetResponder={()=>true}><Text style={s.drawerTitle}>Add company expense</Text><View style={s.methodRow}>{(['Petrol','Food','Travel','Other'] as const).map(x=><Pill key={x} text={x} active={type===x} onPress={()=>setType(x)}/>)}</View><Input label="Amount" value={amount} onChangeText={setAmount} placeholder="₹ 0" keyboardType="numeric"/><Input label="Note" value={note} onChangeText={setNote} placeholder="e.g. train ticket"/><Button title="Save expense" onPress={save}/></View></Pressable></Modal>}
